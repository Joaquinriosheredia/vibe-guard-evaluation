 639              _archive_asset_key = (

 642              # fix(#1290): BEFORE the upsert, not after.
 643              # `create_raster_dataset` reserved only the COG; the kept original

 651                  from app.modules.quota.service import reserve_storage_bytes

 653                  await reserve_storage_bytes(session, uuid.UUID(user_id), archived_bytes)

 655                  session,
 656                  dataset_id=dataset.id,

 664              # REMED-02 / ingest-audit P2-07: stamp terminal progress
 665              # alongside status. ``rows_processed`` stays NULL — raster

 670                  session,
 671                  job_uuid,

 677                  session,
 678                  job_uuid,

 681                      "dataset_id": dataset.id,

 688                  await session.commit()

 690                  # fix(#1778): the one await on this path whose outcome is
 691                  # genuinely unknown — see `publish_commit_landed`. A lost

 727                  "raster_post_publish_followup_failed",

 741          #

 745          try:

 763                  write_failure, job_id=job_id, task="ingest_raster"
 764              )
 765          finally:
 766              # fix(#1213, #1950): the `finally` reapers gate on THIS
 767              # variable, so every exit from this handler sets it — the bounded

 769              final_status = "failed"
 770          # EVENT-03: notify on ingest failed (non-fatal, after commit — deferred import).
 771          # status="failed" is already committed above (err_session.commit) so a
 772          # notification error cannot roll back or alter the terminal job write

 778              build_event_notification,

 791          )
 792          raise
 793      finally:
 794          async with cleanup_step("ingest_raster heartbeat", job_id=job_id):
 795              await stop_ingest_job_heartbeat(heartbeat_task)
 796          # GAP-017: orphan-asset cleanup. If we wrote COG/quicklook bytes to

 806          # Clean up temp COG dir
 807          async with cleanup_step("ingest_raster temp dir", job_id=job_id):

 810          # Clean up local staging file.
 811          # fix(#1290): when `file_path != original_file_path` this is
 812          # a scratch copy downloaded from object storage — always safe to
 813          # remove. When equal, this IS the durable original (local-mode

 818          async with cleanup_step("ingest_raster local file", job_id=job_id):
 819              if file_path != original_file_path:
 820                  _Path(file_path).unlink(missing_ok=True)
 821              elif final_status == "complete" and (
 822                  source_preserved_in_cog or lossy_original_archived
 823              ):
 824                  _Path(file_path).unlink(missing_ok=True)
 825          # fix(#1202): sweep the presigned staging key — raster
 826          # had no equivalent of the vector tail's #430 BA-09 block, so
 827          # nothing on this path ever deleted an object the client could

 829          async with cleanup_step(
 830              "ingest_raster presigned staging object", job_id=job_id
 831          ):
 832              await reap_presigned_staging_object(
 833                  job_id, owned_staging_key, final_status=final_status
 834              )
 835          # fix(#1210), ADR-002 Decision 7: the pre-conversion source object.
 836          # Raster's version of the vector tail's #430 BA-09 block.
 837          # `final_status == "complete"` is reached only after the COG is
 838          # written, read, and committed, so the delete can't race the
 839          # verification it depends on.
 840          #
 841          # fix(#1290): "losslessly" is a claim about the profile that
 842          # ran — under JPEG/WEBP the COG discards detail the upload

 846          #

 848          # a failed conversion leaves those bytes as the operator's only
 849          # diagnostic copy.
 850          async with cleanup_step("ingest_raster downloaded source", job_id=job_id):
 851              if source_preserved_in_cog or lossy_original_archived:

 853                      job_id,
 854                      original_file_path=original_file_path,

 856                      failed_source_replayable=True,
