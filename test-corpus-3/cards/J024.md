 105              log.warn("Delivery {} references missing endpoint {}; marking FAILED",
 106                      deliveryId, delivery.getEndpointId());
 107              delivery.setStatus(DeliveryStatus.FAILED);
 108              delivery.setLastError("endpoint no longer exists");
 109              delivery.setCompletedAt(Instant.now());
 110              deliveryRepository.save(delivery);
 111              return;
 112          }

 115          delivery.setAttempts(delivery.getAttempts() + 1);
 116          // Credentials are decrypted at delivery time only; never logged, never persisted on the row.
 117          String secret = endpointService.decryptSecret(endpoint);
 118          String authorization = endpointService.decryptAuthorizationHeader(endpoint);
 119          try {

 121                      .uri(URI.create(endpoint.getUrl()))
 122                      .contentType(MediaType.APPLICATION_JSON)
 123                      .header(EventHookSignatureService.EVENT_HEADER, delivery.getEventType())
 124                      .header(EventHookSignatureService.DELIVERY_HEADER, delivery.getDeliveryUuid())
 125                      .headers(h -> {

 127                          String customHeadersJson = endpointService.decryptCustomHeaders(endpoint);
 128                          if (customHeadersJson != null && !customHeadersJson.isBlank()) {
 129                              endpoint.parsedCustomHeaders(objectMapper, customHeadersJson).forEach(h::add);

 131                          // Static Authorization wins over any conflicting custom header.
 132                          if (authorization != null) {

 141              }
 142              ResponseEntity<Void> response = request.retrieve().toBodilessEntity();
 143              delivery.setStatus(DeliveryStatus.SUCCESS);
 144              delivery.setLastResponseCode(response.getStatusCode().value());
 145              delivery.setCompletedAt(Instant.now());
